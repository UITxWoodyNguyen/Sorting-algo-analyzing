cmake_minimum_required(VERSION 3.16)

# Copy this file to CMakeLists.txt when starting a new project with the same
# source layout and MinGW g++ compiler convention.

if (NOT DEFINED CMAKE_CXX_COMPILER)
    if (WIN32 AND EXISTS "C:/msys64/mingw64/bin/g++.exe")
        set(CMAKE_CXX_COMPILER "C:/msys64/mingw64/bin/g++.exe"
            CACHE FILEPATH "GNU C++ compiler")
    elseif (UNIX AND EXISTS "/usr/bin/g++")
        set(CMAKE_CXX_COMPILER "/usr/bin/g++"
            CACHE FILEPATH "GNU C++ compiler")
    endif()
endif()

project(SortingAlgoAnalyzingTemplate LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if (MSVC)
    message(FATAL_ERROR
        "Use MinGW g++: C:/msys64/mingw64/bin/g++.exe")
endif()

# Shared include configuration for targets that use src/ headers.
add_library(sort_core INTERFACE)
target_include_directories(sort_core INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/src")

# Main application. Add or remove source files to match the project.
add_executable(sort_app
    src/main.cpp
    src/bench.cpp
    src/kat.cpp
)
target_link_libraries(sort_app PRIVATE sort_core)

# Optional standalone tools.
if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/bench_main.cpp")
    add_executable(bench src/bench.cpp src/bench_main.cpp)
    target_link_libraries(bench PRIVATE sort_core)
endif()

if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/kat_main.cpp")
    add_executable(kat src/kat.cpp src/kat_main.cpp)
    target_link_libraries(kat PRIVATE sort_core)
endif()

if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dataset/generator.cpp")
    add_executable(generator dataset/generator.cpp)
endif()

if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target sort_app bench kat generator)
        if (TARGET ${target})
            target_compile_options(${target} PRIVATE
                -Wall
                -Wextra
                -pedantic
            )
        endif()
    endforeach()
endif()
